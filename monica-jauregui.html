        let audioCtx = null;

        let ambientDrone = null;

        let heartbeatTimer = null;

        let audioEnabled = false;

        function initAudio() {

            if (audioCtx) return;

            try {

                const AudioContext = window.AudioContext || window.webkitAudioContext;

                audioCtx = new AudioContext();

            } catch (e) {

                console.warn("Audio no soportado.");

            }

        }

        function playSynthSound(freq, type, duration, volume = 0.1) {

            if (!audioCtx || !audioEnabled) return;

            try {

                if (audioCtx.state === 'suspended') {

                    audioCtx.resume();

                }

                const osc = audioCtx.createOscillator();

                const gain = audioCtx.createGain();

                osc.type = type;

                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

                gain.gain.setValueAtTime(volume, audioCtx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);

                osc.connect(gain);

                gain.connect(audioCtx.destination);

                osc.start();

                osc.stop(audioCtx.currentTime + duration);

            } catch (e) {}

        }

        function startAmbientDrone() {

            if (!audioCtx || !audioEnabled) return;

            try {

                const osc = audioCtx.createOscillator();

                const filter = audioCtx.createBiquadFilter();

                const gain = audioCtx.createGain();

                osc.type = 'sawtooth';

                osc.frequency.setValueAtTime(55, audioCtx.currentTime);

                filter.type = 'lowpass';

                filter.frequency.setValueAtTime(110, audioCtx.currentTime);

                gain.gain.setValueAtTime(0.06, audioCtx.currentTime);

                osc.connect(filter);

                filter.connect(gain);

                gain.connect(audioCtx.destination);

                osc.start();

                ambientDrone = { osc, gain };

            } catch (e) {}

        }

        function stopAmbientDrone() {

            if (ambientDrone) {

                try { ambientDrone.osc.stop(); } catch(e){}

                ambientDrone = null;

            }

        }

        function updateHeartbeat() {

            if (!audioCtx || !audioEnabled) return;

            if (heartbeatTimer) clearTimeout(heartbeatTimer);

            const distancePercent = gameState.pursuerDistance / 100;

            const interval = 250 + (distancePercent * 1050);

            const rateInSeconds = (interval / 1000).toFixed(2) + 's';

            document.documentElement.style.setProperty('--heart-rate', rateInSeconds);

            const playBeat = () => {

                if (gameState.active && audioEnabled) {

                    playSynthSound(60, 'sine', 0.15, 0.22);

                    setTimeout(() => {

                        if (gameState.active && audioEnabled) {

                            playSynthSound(45, 'sine', 0.2, 0.18);

                        }

                    }, 140);

                }

                heartbeatTimer = setTimeout(updateHeartbeat, interval);

            };

            playBeat();

        }

        function stopHeartbeat() {

            if (heartbeatTimer) {

                clearTimeout(heartbeatTimer);

                heartbeatTimer = null;

            }

        }

        function playScreamJumpscare() {

            if (!audioCtx || !audioEnabled) return;

            try {

                const frequencies = [180, 290, 420, 680, 1050, 1400];

                frequencies.forEach(f => {

                    playSynthSound(f, 'sawtooth', 1.6, 0.25);

                });

               

                const duration = 1.8;

                const bufferSize = audioCtx.sampleRate * duration;

                const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

                const data = buffer.getChannelData(0);

                for (let i = 0; i < bufferSize; i++) {

                    data[i] = Math.random() * 2 - 1;

                }

                const noise = audioCtx.createBufferSource();

                noise.buffer = buffer;

                const noiseGain = audioCtx.createGain();

                noiseGain.gain.setValueAtTime(0.35, audioCtx.currentTime);

                noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);

                noise.connect(noiseGain);

                noiseGain.connect(audioCtx.destination);

                noise.start();

            } catch(e) {}

        }

        function shuffleArray(array) {

            const arr = [...array];

            for (let i = arr.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

                [arr[i], arr[j]] = [arr[j], arr[i]];

            }

            return arr;

        }

        function startGame(withAudio) {

            audioEnabled = withAudio;

                initAudio();

                startAmbientDrone();

            document.getElementById('screenIntro').classList.add('hidden');

            document.getElementById('screenGame').classList.remove('hidden');

            gameState.active = true;

            gameState.pursuerDistance = 100;

            gameState.correctAnswers = 0;

            gameState.roomIndex = 0;

            gameState.shuffledQuestions = shuffleArray(riddleDatabase);

           

            updateUI();

            nextQuestion();

            // Sombra avanza paulatinamente en base al tiempo

            gameState.naturalPursuitInterval = setInterval(() => {

                if (gameState.active) {

                    const difficultyFactor = 1 + (gameState.roomIndex * 0.15);

                    const decrease = Number((1.1 * difficultyFactor).toFixed(1));

                    modifyDistance(-decrease);

                }

            }, 1200);

            if (audioEnabled) {

                updateHeartbeat();

            }

        }

        function modifyDistance(amount) {

            if (!gameState.active) return;

            gameState.pursuerDistance += amount;

            if (gameState.pursuerDistance > 100) gameState.pursuerDistance = 100;

            const calculatedOpacity = gameState.pursuerDistance < 50 ? (50 - gameState.pursuerDistance) / 50 * 0.7 : 0;

            document.getElementById('vignette').style.setProperty('--danger-opacity', calculatedOpacity);

            if (gameState.pursuerDistance <= 0) {

                gameState.pursuerDistance = 0;

                triggerJumpscare();

            }

            updateUI();

        }

        function updateUI() {

            document.getElementById('distanceText').innerText = `DISTANCIA: ${Math.max(0, Math.floor(gameState.pursuerDistance))}m`;

            const distanceBar = document.getElementById('distanceBar');

            distanceBar.style.width = `${gameState.pursuerDistance}%`;

           

            if (gameState.pursuerDistance > 55) {

                distanceBar.className = "bg-gradient-to-r from-emerald-800 to-green-500 h-full w-full transition-all duration-300";

            } else if (gameState.pursuerDistance > 25) {

                distanceBar.className = "bg-gradient-to-r from-amber-700 to-yellow-500 h-full w-full transition-all duration-300";

            } else {

                distanceBar.className = "bg-gradient-to-r from-red-900 to-red-600 h-full w-full transition-all duration-300";

            }

            document.getElementById('progressText').innerText = `Sala: ${gameState.roomIndex} / ${gameState.maxRooms}`;

        }

        function nextQuestion() {

            if (!gameState.active) return;

            if (gameState.roomIndex >= gameState.maxRooms) {

                triggerVictory();

                return;

            }

            if (gameState.shuffledQuestions.length === 0) {

                gameState.shuffledQuestions = shuffleArray(riddleDatabase);

            }

            gameState.currentQuestion = gameState.shuffledQuestions.pop();

            document.getElementById('questionCategory').innerText = gameState.currentQuestion.category;

            document.getElementById('questionText').innerText = gameState.currentQuestion.text;

            const container = document.getElementById('optionsContainer');

            container.innerHTML = '';

           

            gameState.currentQuestion.options.forEach((opt, idx) => {

                const btn = document.createElement('button');

                btn.className = "w-full text-left bg-zinc-900 hover:bg-zinc-800 active:bg-zinc-700 border border-zinc-800 hover:border-red-900 text-zinc-300 hover:text-white transition-all duration-200 p-4 rounded-xl text-sm md:text-base font-medium flex items-center gap-3";

            });

            gameState.timerRemaining = gameState.timerLimit;

           

            document.getElementById('timerSeconds').innerText = `${gameState.timerRemaining}s`;

            document.getElementById('timerBar').style.width = '100%';

                if (gameState.active) {

                    gameState.timerRemaining--;

                    document.getElementById('timerSeconds').innerText = `${gameState.timerRemaining}s`;

                   

                    const percent = (gameState.timerRemaining / gameState.timerLimit) * 100;

                    document.getElementById('timerBar').style.width = `${percent}%`;

                    if (gameState.timerRemaining <= 5) {

                        document.getElementById('timerBar').className = "absolute top-0 left-0 h-1 bg-red-600 transition-all duration-100 ease-linear";

                    }

                        clearInterval(gameState.gameTimerInterval);

                        handleTimeOut();

                    }

                }

            }, 1000);

        }

        function handleTimeOut() {

            if (!gameState.active) return;

            if (audioEnabled) playSynthSound(100, 'sawtooth', 0.6, 0.2);

            showFeedback("¡TIEMPO AGOTADO! La Sombra avanza veloz.", false);

            modifyDistance(-25);

            setTimeout(() => { if (gameState.active) nextQuestion(); }, 1600);

        }

        function submitAnswer(selectedIndex) {

            if (!gameState.active) return;

            clearInterval(gameState.gameTimerInterval);

            const buttons = document.getElementById('optionsContainer').getElementsByTagName('button');

            for(let b of buttons) b.disabled = true;

            const isCorrect = selectedIndex === gameState.currentQuestion.correct;

            if (isCorrect) {

                gameState.correctAnswers++;

                gameState.roomIndex++;

                if (audioEnabled) {

                    playSynthSound(330, 'sine', 0.1, 0.12);

                    setTimeout(() => playSynthSound(440, 'sine', 0.2, 0.12), 80);

                }

                showFeedback("Correcto. Has ganado algo de ventaja.", true);

                modifyDistance(15);

            } else {

                if (audioEnabled) playSynthSound(130, 'sawtooth', 0.5, 0.2);

                showFeedback("Incorrecto. Escuchas pasos acelerados...", false);

                modifyDistance(-20);

            }

            setTimeout(() => { if (gameState.active) nextQuestion(); }, 1800);

        }

        }
